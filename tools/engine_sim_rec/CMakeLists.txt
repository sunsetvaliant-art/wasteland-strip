cmake_minimum_required(VERSION 3.10)
project(es_rec CXX)
set(CMAKE_CXX_STANDARD 17)
add_compile_definitions(__forceinline=inline\ __attribute__\(\(always_inline\)\))
add_compile_options(-O2 -w -fdelayed-template-parsing -fms-extensions "SHELL:-include ${CMAKE_CURRENT_SOURCE_DIR}/prelude.h")
set(CMAKE_BUILD_TYPE Release)
set(ES ${CMAKE_CURRENT_SOURCE_DIR}/engine-sim)
add_compile_definitions(ATG_ENGINE_SIM_PIRANHA_ENABLED)
set(SUB ${ES}/dependencies/submodules)
file(GLOB SCS_SRC ${SUB}/simple-2d-constraint-solver/src/*.cpp)
add_library(simple-2d-constraint-solver STATIC ${SCS_SRC})
add_library(csv-io STATIC ${SUB}/csv-io/src/csv_data.cpp ${SUB}/csv-io/src/information.cpp)
find_package(FLEX REQUIRED)
find_package(BISON REQUIRED)
find_package(Boost COMPONENTS filesystem REQUIRED)
flex_target(lexer ${SUB}/piranha/flex-bison/scanner.l ${CMAKE_CURRENT_BINARY_DIR}/scanner.auto.cpp DEFINES_FILE ${CMAKE_CURRENT_BINARY_DIR}/scanner.auto.h)
bison_target(parser ${SUB}/piranha/flex-bison/specification.y ${CMAKE_CURRENT_BINARY_DIR}/parser.auto.cpp DEFINES_FILE ${CMAKE_CURRENT_BINARY_DIR}/parser.auto.h)
add_flex_bison_dependency(lexer parser)
file(GLOB PIR_SRC ${SUB}/piranha/src/*.cpp)
add_library(piranha STATIC ${PIR_SRC} ${BISON_parser_OUTPUTS} ${FLEX_lexer_OUTPUTS})
target_include_directories(piranha PUBLIC ${CMAKE_CURRENT_BINARY_DIR} ${SUB}/piranha/dependencies/libraries/flex/include ${SUB}/piranha/include)
target_link_libraries(piranha Boost::filesystem)
set(CORE ${ES}/src/audio_buffer.cpp ${ES}/src/camshaft.cpp ${ES}/src/crankshaft.cpp ${ES}/src/combustion_chamber.cpp ${ES}/src/connecting_rod.cpp ${ES}/src/convolution_filter.cpp ${ES}/src/cylinder_bank.cpp ${ES}/src/cylinder_head.cpp ${ES}/src/delay_filter.cpp ${ES}/src/derivative_filter.cpp ${ES}/src/direct_throttle_linkage.cpp ${ES}/src/dynamometer.cpp ${ES}/src/engine.cpp ${ES}/src/exhaust_system.cpp ${ES}/src/feedback_comb_filter.cpp ${ES}/src/filter.cpp ${ES}/src/fuel.cpp ${ES}/src/function.cpp ${ES}/src/gas_system.cpp ${ES}/src/gaussian_filter.cpp ${ES}/src/governor.cpp ${ES}/src/ignition_module.cpp ${ES}/src/impulse_response.cpp ${ES}/src/intake.cpp ${ES}/src/jitter_filter.cpp ${ES}/src/leveling_filter.cpp ${ES}/src/low_pass_filter.cpp ${ES}/src/part.cpp ${ES}/src/piston.cpp ${ES}/src/piston_engine_simulator.cpp ${ES}/src/simulator.cpp ${ES}/src/standard_valvetrain.cpp ${ES}/src/starter_motor.cpp ${ES}/src/synthesizer.cpp ${ES}/src/throttle.cpp ${ES}/src/transmission.cpp ${ES}/src/utilities.cpp ${ES}/src/valvetrain.cpp ${ES}/src/vehicle.cpp ${ES}/src/vehicle_drag_constraint.cpp ${ES}/src/vtec_valvetrain.cpp)
add_library(core STATIC ${CORE})
target_include_directories(core PUBLIC ${ES}/dependencies/submodules)
target_link_libraries(core simple-2d-constraint-solver csv-io)
add_library(script STATIC ${ES}/scripting/src/channel_types.cpp ${ES}/scripting/src/compiler.cpp ${ES}/scripting/src/engine_context.cpp ${ES}/scripting/src/language_rules.cpp)
target_include_directories(script PUBLIC ${ES}/dependencies/submodules)
target_link_libraries(script core csv-io piranha)
add_executable(es_rec rec.cpp)
target_link_libraries(es_rec script core pthread "-Wl,--allow-multiple-definition")
